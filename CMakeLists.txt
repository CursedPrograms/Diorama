cmake_minimum_required(VERSION 3.20)
project(Diorama VERSION 0.1.0 LANGUAGES C CXX)
# Diorama: little worlds on AR markers, laid out like Unity; exports jsartoolkit5 web packages. Part of Atelier.
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

include(FetchContent)
# Dear ImGui's docking branch: the editor's windows resize, move, snap and tab like Unity's
if(EXISTS ${CMAKE_SOURCE_DIR}/external/imgui-1.90.4-docking/imgui.h)
    set(FETCHCONTENT_SOURCE_DIR_IMGUI ${CMAKE_SOURCE_DIR}/external/imgui-1.90.4-docking)
endif()
FetchContent_Declare(imgui URL https://github.com/ocornut/imgui/archive/refs/tags/v1.90.4-docking.tar.gz)
FetchContent_MakeAvailable(imgui)
# QuickJS (quickjs-ng, MIT): the native player's scripts
if(EXISTS ${CMAKE_SOURCE_DIR}/external/quickjs/quickjs.h)
    set(FETCHCONTENT_SOURCE_DIR_QUICKJS ${CMAKE_SOURCE_DIR}/external/quickjs)
endif()
FetchContent_Declare(quickjs URL https://github.com/quickjs-ng/quickjs/archive/refs/tags/v0.10.1.tar.gz EXCLUDE_FROM_ALL)
FetchContent_MakeAvailable(quickjs)
# Jolt Physics (MIT): the editor's Simulate and the native player; the web game runs the same engine as WASM
# (assets/web/jolt: JoltPhysics.js 1.1.1, Jolt 5.6). Built for any x64 processor (SSE2), without its samples and tests.
foreach(o USE_SSE4_1 USE_SSE4_2 USE_AVX USE_AVX2 USE_LZCNT USE_TZCNT USE_F16C USE_FMADD ENABLE_ALL_WARNINGS ENABLE_INSTALL
          TARGET_UNIT_TESTS TARGET_HELLO_WORLD TARGET_PERFORMANCE_TEST TARGET_SAMPLES TARGET_VIEWER INTERPROCEDURAL_OPTIMIZATION
          DEBUG_RENDERER_IN_DEBUG_AND_RELEASE PROFILER_IN_DEBUG_AND_RELEASE ENABLE_OBJECT_STREAM JPH_USE_DX12 JPH_USE_VK JPH_USE_MTL)
    set(${o} OFF CACHE BOOL "" FORCE)
endforeach()
set(USE_STATIC_MSVC_RUNTIME_LIBRARY ON CACHE BOOL "" FORCE)
if(EXISTS ${CMAKE_SOURCE_DIR}/external/JoltPhysics-5.6.0/Jolt/Jolt.h)
    set(FETCHCONTENT_SOURCE_DIR_JOLT ${CMAKE_SOURCE_DIR}/external/JoltPhysics-5.6.0)
endif()
FetchContent_Declare(jolt URL https://github.com/jrouwe/JoltPhysics/archive/refs/tags/v5.6.0.tar.gz SOURCE_SUBDIR Build EXCLUDE_FROM_ALL)
FetchContent_MakeAvailable(jolt)
if(WIN32 AND NOT SDL2_DIR AND EXISTS ${CMAKE_SOURCE_DIR}/external/SDL2-2.30.9/cmake)
    set(SDL2_DIR ${CMAKE_SOURCE_DIR}/external/SDL2-2.30.9/cmake)
endif()
find_package(SDL2 REQUIRED)
find_package(OpenGL REQUIRED)
# single-file libraries: stb (images, fonts), cgltf (glTF models)
set(STB_DIR ${CMAKE_SOURCE_DIR}/external/stb)
foreach(f stb_image.h stb_image_write.h stb_truetype.h)
    if(NOT EXISTS ${STB_DIR}/${f})
        file(DOWNLOAD https://raw.githubusercontent.com/nothings/stb/master/${f} ${STB_DIR}/${f})
    endif()
endforeach()
if(NOT EXISTS ${CMAKE_SOURCE_DIR}/external/cgltf.h)
    file(DOWNLOAD https://raw.githubusercontent.com/jkuhlmann/cgltf/master/cgltf.h ${CMAKE_SOURCE_DIR}/external/cgltf.h)
endif()
add_library(imgui STATIC
    ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp
    ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_sdl2.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
target_include_directories(imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
target_link_libraries(imgui PUBLIC SDL2::SDL2 OpenGL::GL)

set(SOURCES src/gui.cpp src/splash.cpp src/scene.cpp src/physics.cpp src/arnav.cpp src/marker.cpp src/model.cpp src/viewer.cpp src/serve.cpp src/stb.cpp external/textedit/TextEditor.cpp)
if(WIN32)
    list(APPEND SOURCES src/app.rc)
endif()
add_executable(Diorama WIN32 ${SOURCES})
target_include_directories(Diorama PRIVATE src ${STB_DIR} ${CMAKE_SOURCE_DIR}/external ${CMAKE_SOURCE_DIR}/external/textedit)
target_link_libraries(Diorama PRIVATE imgui Jolt)
target_compile_definitions(Diorama PRIVATE DIORAMA_VERSION="${PROJECT_VERSION}")
if(TARGET SDL2::SDL2main)
    target_link_libraries(Diorama PRIVATE SDL2::SDL2main)
endif()
if(MSVC)
    target_compile_options(Diorama PRIVATE /W3 /utf-8 /O2 /D_CRT_SECURE_NO_WARNINGS)
endif()
add_custom_command(TARGET Diorama POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/assets $<TARGET_FILE_DIR:Diorama>/assets
    COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/templates $<TARGET_FILE_DIR:Diorama>/templates)
if(WIN32)
    add_custom_command(TARGET Diorama POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL2::SDL2> $<TARGET_FILE_DIR:Diorama>)
endif()

# ---- DioramaGame.exe: the Windows player (a built game as a Windows program: WebView2 + a local server) ----
if(WIN32)
    set(WV2_DIR ${CMAKE_SOURCE_DIR}/external/webview2)
    if(NOT EXISTS ${WV2_DIR}/build/native/include/WebView2.h)
        file(DOWNLOAD https://www.nuget.org/api/v2/package/Microsoft.Web.WebView2/1.0.4314.35 ${CMAKE_BINARY_DIR}/webview2.zip)
        file(ARCHIVE_EXTRACT INPUT ${CMAKE_BINARY_DIR}/webview2.zip DESTINATION ${WV2_DIR})
    endif()
    add_executable(DioramaGame WIN32 player/win/main.cpp player/win/app.rc)
    target_include_directories(DioramaGame PRIVATE ${WV2_DIR}/build/native/include)
    target_link_libraries(DioramaGame PRIVATE ${WV2_DIR}/build/native/x64/WebView2LoaderStatic.lib ws2_32 shlwapi)
    target_compile_definitions(DioramaGame PRIVATE UNICODE _UNICODE)
    if(MSVC)
        target_compile_options(DioramaGame PRIVATE /utf-8 /O2 /D_CRT_SECURE_NO_WARNINGS)
    endif()
    # beside Diorama.exe (in its player folder): Diorama copies it into Windows builds
    set_target_properties(DioramaGame PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE ${CMAKE_BINARY_DIR}/Release/player RUNTIME_OUTPUT_DIRECTORY_DEBUG ${CMAKE_BINARY_DIR}/Debug/player)
    add_dependencies(Diorama DioramaGame)
endif()

# ---- OpenCV (optional): the webcam surface tracker for the native player, and its test tool ----
if(NOT OpenCV_DIR AND EXISTS "D:/cc/_src/opencv-dl/opencv/build/OpenCVConfig.cmake")
    set(OpenCV_DIR "D:/cc/_src/opencv-dl/opencv/build")
endif()
find_package(OpenCV CONFIG QUIET)
if(OpenCV_FOUND)
    add_library(surfacetrack STATIC src/surfacetrack.cpp src/arucotrack.cpp src/calibrate.cpp src/markertrack.cpp)   # the native trackers (surfaces, ArUco)
    target_include_directories(surfacetrack PUBLIC src ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(surfacetrack PUBLIC ${OpenCV_LIBS})
    add_executable(surfacetrack-tool tools/surfacetrack/main.cpp)
    set_target_properties(surfacetrack-tool PROPERTIES OUTPUT_NAME surfacetrack RUNTIME_OUTPUT_DIRECTORY_RELEASE ${CMAKE_BINARY_DIR}/Release/tools)
    target_link_libraries(surfacetrack-tool PRIVATE surfacetrack)
    if(MSVC)
        target_compile_options(surfacetrack PRIVATE /O2 /utf-8)
        target_compile_options(surfacetrack-tool PRIVATE /O2 /utf-8)
    endif()
    if(WIN32)
        add_custom_command(TARGET surfacetrack-tool POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${OpenCV_DIR}/x64/vc16/bin/opencv_world4100.dll" "${OpenCV_DIR}/x64/vc16/bin/opencv_videoio_ffmpeg4100_64.dll" $<TARGET_FILE_DIR:surfacetrack-tool>)
    endif()
    add_executable(arucotrack-tool tools/arucotrack/main.cpp)
    set_target_properties(arucotrack-tool PROPERTIES OUTPUT_NAME arucotrack RUNTIME_OUTPUT_DIRECTORY_RELEASE ${CMAKE_BINARY_DIR}/Release/tools)
    target_link_libraries(arucotrack-tool PRIVATE surfacetrack)
    add_executable(calibrate-tool tools/calibrate/main.cpp)
    set_target_properties(calibrate-tool PROPERTIES OUTPUT_NAME calibrate RUNTIME_OUTPUT_DIRECTORY_RELEASE ${CMAKE_BINARY_DIR}/Release/tools)
    target_link_libraries(calibrate-tool PRIVATE surfacetrack)
    add_executable(markertrack-tool tools/markertrack/main.cpp)
    set_target_properties(markertrack-tool PROPERTIES OUTPUT_NAME markertrack RUNTIME_OUTPUT_DIRECTORY_RELEASE ${CMAKE_BINARY_DIR}/Release/tools)
    target_link_libraries(markertrack-tool PRIVATE surfacetrack)
    add_executable(physicstest tools/physicstest/main.cpp src/physics.cpp src/scene.cpp src/model.cpp src/stb.cpp)   # the physics, measured (and test.js: the web's)
    set_target_properties(physicstest PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE ${CMAKE_BINARY_DIR}/Release/tools)
    target_include_directories(physicstest PRIVATE src ${STB_DIR} ${CMAKE_SOURCE_DIR}/external)
    target_link_libraries(physicstest PRIVATE Jolt)
    if(MSVC)
        target_compile_options(physicstest PRIVATE /utf-8 /O2 /D_CRT_SECURE_NO_WARNINGS)
    endif()
    add_executable(gen_aruco tools/gen_aruco/main.cpp)   # writes src/aruco_dicts.h (committed: the editor needs no OpenCV)
    set_target_properties(gen_aruco PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE ${CMAKE_BINARY_DIR}/Release/tools)
    target_link_libraries(gen_aruco PRIVATE ${OpenCV_LIBS})
    target_include_directories(gen_aruco PRIVATE ${OpenCV_INCLUDE_DIRS})
    # ---- SLAM (stella_vslam, BSD-2: built static with D:/cc/_src/slam/build_deps.sh + build_vslam.sh), optional
    if(NOT DIORAMA_SLAM_PREFIX AND EXISTS "D:/cc/_src/slam/install/lib/cmake/stella_vslam")
        set(DIORAMA_SLAM_PREFIX "D:/cc/_src/slam/install")
    endif()
    if(DIORAMA_SLAM_PREFIX)
        list(APPEND CMAKE_PREFIX_PATH ${DIORAMA_SLAM_PREFIX})
        set(Eigen3_DIR ${DIORAMA_SLAM_PREFIX}/share/eigen3/cmake)
        find_package(stella_vslam CONFIG QUIET)
    endif()
    if(stella_vslam_FOUND)
        message(STATUS "stella_vslam: SLAM is built")
        add_executable(slamtest tools/slamtest/main.cpp src/slam.cpp)
        target_include_directories(slamtest PRIVATE src)
        set_target_properties(slamtest PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE ${CMAKE_BINARY_DIR}/Release/tools)
        target_link_libraries(slamtest PRIVATE stella_vslam::stella_vslam)
        target_compile_definitions(slamtest PRIVATE DIORAMA_SLAM _USE_MATH_DEFINES NOMINMAX)
        target_compile_options(slamtest PRIVATE /utf-8 /bigobj)
    endif()
    # ---- DioramaNative.exe: the native player (Diorama's renderer, the C++ trackers, the PC's camera)
    # its script runtime (player/native/runtime.js), inside the exe as bytes
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/player/native/runtime.js)
    file(READ ${CMAKE_SOURCE_DIR}/player/native/runtime.js RUNTIME_HEX HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," RUNTIME_HEX "${RUNTIME_HEX}")
    file(WRITE ${CMAKE_BINARY_DIR}/gen/runtime_js.h "// made by CMake from player/native/runtime.js\nstatic const unsigned char RUNTIME_JS[] = {${RUNTIME_HEX}};\n")
    add_executable(DioramaNative WIN32 player/native/main.cpp player/native/script.cpp player/native/ui.cpp src/scene.cpp src/physics.cpp src/arnav.cpp src/model.cpp src/viewer.cpp src/stb.cpp src/slam.cpp player/win/app.rc)
    target_include_directories(DioramaNative PRIVATE src player/native ${CMAKE_BINARY_DIR}/gen ${STB_DIR} ${CMAKE_SOURCE_DIR}/external)
    target_link_libraries(DioramaNative PRIVATE surfacetrack imgui qjs Jolt)
    if(stella_vslam_FOUND)   # SLAM in the player (else src/slam.cpp says it isn't there)
        target_link_libraries(DioramaNative PRIVATE stella_vslam::stella_vslam)
        target_compile_definitions(DioramaNative PRIVATE DIORAMA_SLAM _USE_MATH_DEFINES NOMINMAX)
        target_compile_options(DioramaNative PRIVATE /bigobj)
    endif()
    if(TARGET SDL2::SDL2main)
        target_link_libraries(DioramaNative PRIVATE SDL2::SDL2main)
    endif()
    if(MSVC)
        target_compile_options(DioramaNative PRIVATE /W3 /utf-8 /O2 /D_CRT_SECURE_NO_WARNINGS)
    endif()
    set_target_properties(DioramaNative PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE ${CMAKE_BINARY_DIR}/Release/player RUNTIME_OUTPUT_DIRECTORY_DEBUG ${CMAKE_BINARY_DIR}/Debug/player)
    add_custom_command(TARGET DioramaNative POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${OpenCV_DIR}/x64/vc16/bin/opencv_world4100.dll" "${OpenCV_DIR}/x64/vc16/bin/opencv_videoio_ffmpeg4100_64.dll" $<TARGET_FILE:SDL2::SDL2> $<TARGET_FILE_DIR:DioramaNative>)
    add_dependencies(Diorama DioramaNative)
    message(STATUS "OpenCV ${OpenCV_VERSION}: the surface tracker is built")
else()
    message(STATUS "No OpenCV: the native surface tracker is left out")
endif()
